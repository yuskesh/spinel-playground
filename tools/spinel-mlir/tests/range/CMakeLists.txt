cmake_minimum_required(VERSION 3.20)
if(CMAKE_SOURCE_DIR STREQUAL CMAKE_CURRENT_SOURCE_DIR)
  project(SpinelGroupARangeTests LANGUAGES C CXX)
  find_package(LLVM 23.1.1 REQUIRED CONFIG)
endif()
add_compile_options(-fno-exceptions -fno-rtti)
add_library(SpinelGroupARange STATIC ../../lib/Analysis/GroupARange.cpp)
target_include_directories(SpinelGroupARange PUBLIC ../../include ${LLVM_INCLUDE_DIRS})
target_compile_features(SpinelGroupARange PUBLIC cxx_std_17)
llvm_map_components_to_libnames(llvm_libs Support)
target_link_libraries(SpinelGroupARange PUBLIC ${llvm_libs})
add_executable(spinel-group-a-range-test GroupARangeTest.cpp)
target_link_libraries(spinel-group-a-range-test PRIVATE SpinelGroupARange)
enable_testing()
add_test(NAME group-a-range COMMAND spinel-group-a-range-test ${CMAKE_CURRENT_SOURCE_DIR}/diagnostics.tsv)
add_executable(spinel-group-a-range-mutation-test RangeMutationTest.cpp)
target_link_libraries(spinel-group-a-range-mutation-test PRIVATE SpinelGroupARange)
add_test(NAME group-a-range-mutations COMMAND spinel-group-a-range-mutation-test)
if(TARGET check-spinel-math)
  add_dependencies(check-spinel-math spinel-group-a-range-test spinel-group-a-range-mutation-test)
endif()
