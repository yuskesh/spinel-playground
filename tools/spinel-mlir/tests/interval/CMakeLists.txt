cmake_minimum_required(VERSION 3.20)
if(CMAKE_SOURCE_DIR STREQUAL CMAKE_CURRENT_SOURCE_DIR)
  project(SpinelMathInterval LANGUAGES C CXX)
  find_package(LLVM 23.1.1 REQUIRED CONFIG)
endif()
add_compile_options(-fno-exceptions -fno-rtti)
add_executable(spinel-math-interval-test IntervalTest.cpp)
target_compile_features(spinel-math-interval-test PRIVATE cxx_std_17)
target_include_directories(spinel-math-interval-test PRIVATE ../../include ${LLVM_INCLUDE_DIRS})
llvm_map_components_to_libnames(llvm_libs Support)
target_link_libraries(spinel-math-interval-test PRIVATE ${llvm_libs})
enable_testing()
add_test(NAME math-interval-oracle COMMAND spinel-math-interval-test ${CMAKE_CURRENT_SOURCE_DIR}/cases.tsv)
foreach(test Domain Mutation)
  add_executable(spinel-math-${test}-test ${test}Test.cpp)
  target_compile_features(spinel-math-${test}-test PRIVATE cxx_std_17)
  target_include_directories(spinel-math-${test}-test PRIVATE ../../include ${LLVM_INCLUDE_DIRS})
  target_link_libraries(spinel-math-${test}-test PRIVATE ${llvm_libs})
  add_test(NAME math-interval-${test} COMMAND spinel-math-${test}-test)
endforeach()
if(TARGET SpinelMathIntervalAnalysis)
  add_llvm_executable(spinel-math-solver-test PARTIAL_SOURCES_INTENDED SolverTest.cpp)
  llvm_update_compile_flags(spinel-math-solver-test)
  target_link_libraries(spinel-math-solver-test PRIVATE SpinelMathIntervalAnalysis MLIRParser MLIRFuncDialect MLIRControlFlowDialect)
  add_test(NAME math-interval-solver COMMAND spinel-math-solver-test)
  add_custom_target(check-spinel-math
    COMMAND ${CMAKE_CTEST_COMMAND} --test-dir ${CMAKE_BINARY_DIR} --output-on-failure
    DEPENDS spinel-math-solver-test spinel-math-interval-test spinel-math-Domain-test spinel-math-Mutation-test USES_TERMINAL)
  add_dependencies(check-spinel-mlir check-spinel-math)
endif()
