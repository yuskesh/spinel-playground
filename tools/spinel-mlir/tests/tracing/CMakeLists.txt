cmake_minimum_required(VERSION 3.20)
project(SpinelTracingTests LANGUAGES C CXX)
find_package(LLVM 23.1.1 REQUIRED CONFIG)
add_compile_options(-fno-exceptions -fno-rtti)
add_executable(spinel-kernel-types-test KernelTypesTest.cpp ../../lib/Analysis/KernelTypes.cpp)
target_include_directories(spinel-kernel-types-test PRIVATE ../../include ${LLVM_INCLUDE_DIRS})
target_compile_features(spinel-kernel-types-test PRIVATE cxx_std_17)
llvm_map_components_to_libnames(llvm_libs Support)
target_link_libraries(spinel-kernel-types-test PRIVATE ${llvm_libs})
enable_testing()
add_test(NAME kernel-types COMMAND spinel-kernel-types-test)
